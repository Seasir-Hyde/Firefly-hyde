## 创建组件

文件路径：`src/components/features/WelcomeToast.astro`

> [!NOTE] 版本说明
> 本节贴出的是**初版实现**（DOM 拼接 + Tailwind 类名）。当前组件已重构为 Web Component 版本（`<welcome-toast>` 自定义元素 + 作用域样式），文件路径为 `src/components/features/WelcomeToast.astro`，详见文末「行为变化」。

```astro
---
// src/components/WelcomeToast.astro
---

<script>
    const VISIT_SESSION_KEY = 'blog_visit_flag';
    let hasShownToast = false;

    function hasVisitedThisSession() {
        try {
            return sessionStorage.getItem(VISIT_SESSION_KEY) === '1';
        } catch {
            return false; // 隐私模式等 sessionStorage 不可用时按未访问处理
        }
    }

    function markVisitedThisSession() {
        try {
            sessionStorage.setItem(VISIT_SESSION_KEY, '1');
        } catch { /* 静默跳过 */ }
    }

    function createWelcomeToast() {
        const toast = document.createElement('div');
        toast.id = 'welcome-toast';
        // Tailwind 样式控制：自动适配深色模式
        toast.className = 'fixed bottom-4 right-4 z-50 transform translate-y-full opacity-0 transition-all duration-500 ease-out';
        
        //使用 bg-white dark:bg-zinc-800 强制适配颜色
        toast.innerHTML = `
            <div class="bg-white dark:bg-zinc-800 border border-gray-200 dark:border-zinc-700 rounded-lg shadow-lg p-4 max-w-xs">
                <div class="flex items-start gap-3">
                    <div class="flex-shrink-0 text-2xl">👋</div>
                    <div class="flex-1 min-w-0">
                        <p id="welcome-message" class="text-sm font-semibold text-gray-900 dark:text-gray-100">
                            正在加载...
                        </p>
                        <p class="text-xs text-gray-600 dark:text-gray-400 mt-2 italic">
                            欢迎来到我的博客
                        </p>
                    </div>
                    <button 
                        onclick="closeWelcomeToast()"
                        class="flex-shrink-0 text-gray-400 hover:text-gray-800 dark:hover:text-gray-200 transition-colors"
                        aria-label="关闭"
                    >
                        <svg class="w-4 h-4" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                            <path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M6 18L18 6M6 6l12 12"></path>
                        </svg>
                    </button>
                </div>
            </div>
        `;
        document.body.appendChild(toast);
        return toast;
    }

    function closeWelcomeToast() {
        const toast = document.getElementById('welcome-toast');
        if (toast) {
            toast.classList.add('translate-y-full', 'opacity-0');
            toast.classList.remove('translate-y-0', 'opacity-100');
            setTimeout(() => { toast.remove(); }, 500);
        }
    }

    async function fetchLocation() {
        const controller = new AbortController();
        const timer = setTimeout(() => controller.abort(), 5000);
        try {
            const response = await fetch('https://v2.xxapi.cn/api/ip', { signal: controller.signal });
            const data = await response.json();
            if (data.code === 200 && data.data) {
                const locationText = `你好，来自${data.data.address}的朋友`;
                const messageEl = document.getElementById('welcome-message');
                if (messageEl) messageEl.textContent = locationText;
            }
        } catch (error) {
            const messageEl = document.getElementById('welcome-message');
            if (messageEl) messageEl.textContent = '你好，欢迎来到我的博客';
        } finally {
            clearTimeout(timer);
        }
    }

    async function showWelcomeToast() {
        if (hasShownToast || hasVisitedThisSession()) return;
        const toast = createWelcomeToast();
        setTimeout(() => {
            toast.classList.remove('translate-y-full', 'opacity-0');
            toast.classList.add('translate-y-0', 'opacity-100');
        }, 100);
        hasShownToast = true;
        markVisitedThisSession();
        // 延迟到浏览器空闲时再请求第三方 IP 定位接口，避免与首屏关键资源抢带宽
        const schedule = window.requestIdleCallback || function (cb) { return setTimeout(cb, 300); };
        schedule(() => {
            // 用户已提前关闭 toast 时跳过第三方请求，避免无效调用
            if (document.getElementById('welcome-toast')) fetchLocation();
        });
        setTimeout(() => { closeWelcomeToast(); }, 5000);
    }

    function initWelcome() {
        // 检查是否为首页（路径为 "/" 或 "/index.html"）
        const isHomePage = window.location.pathname === '/' || window.location.pathname === '/index.html';
        
        // 只在首页显示欢迎弹窗
        if (isHomePage) {
            showWelcomeToast();
        }
    }

    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', initWelcome);
        document.addEventListener('astro:page-load', initWelcome);
    } else {
        initWelcome();
    }

    (window as any).closeWelcomeToast = closeWelcomeToast;
</script>

<style>
    /* 样式保持不变 */
    #welcome-toast { max-width: calc(100vw - 2rem); }
    @media (max-width: 640px) {
        #welcome-toast {
            top: auto; bottom: 1rem; left: 50%; right: auto;
            transform: translateX(-50%) translateY(100%);
            width: 90%; max-width: none;
            transition: transform 0.5s ease-out, opacity 0.5s ease-out;
        }
        #welcome-toast.translate-y-0 { transform: translateX(-50%) translateY(0); }
        #welcome-toast.translate-y-full { transform: translateX(-50%) translateY(100%); }
    }
</style>
```

## 显示逻辑配置

- 访客只在首次访问时显示欢迎弹窗（原作者）

```ts
    function initWelcome() {
        // 检查是否已显示过欢迎弹窗
        if (!sessionStorage.getItem(VISIT_SESSION_KEY)) {
            sessionStorage.setItem(VISIT_SESSION_KEY, 'true');
            showWelcomeToast(); // 显示欢迎弹窗
        }
    }
```

- 访客只会在首页刷新时显示，非首页刷新不会显示（正在使用）

```ts {3}
    function initWelcome() {
        // 检查是否为首页（路径为 "/" 或 "/index.html"）
        const isHomePage = window.location.pathname === '/' || window.location.pathname === '/index.html';
        
        // 只在首页显示欢迎弹窗
        if (isHomePage) {
            showWelcomeToast();
        }
    }
```

- 访客每次刷新页面时，欢迎提示组件会重新显示，不限于文章页。

```ts {3}
    function initWelcome() {
        // 显示欢迎弹窗
        showWelcomeToast();
    }
```

## 行为变化
- 26.10.03
- 首页首次访问：正常弹窗，空闲时请求定位 API — 与之前一致
- 同会话内整页刷新首页： 不再重复弹窗 （新行为）
- 非首页：不再下载/执行 WelcomeToast 脚本（新行为）
- 组件内`pathname` 判断保留，作为兜底
- 26.11.02
- 组件重构：替换为 Web Component 实现（`<welcome-toast>` 自定义元素 + 作用域样式 + `data-state` 状态机），替代原先的 DOM 拼接方案
- 文件路径迁移：`src/components/widget/WelcomeToast.astro` → `src/components/features/WelcomeToast.astro`
- 交互变化：入场动画改为 `translateY + scale` 组合过渡；新增关闭按钮的悬停旋转反馈；展示时长 5s → 6.5s；延迟 850ms 后弹出
- 定位接口改由 `serviceConfig.visitorLocationUrl` 配置（默认 `https://v2.xxapi.cn/api/ip`），留空则只显示默认欢迎语
- 会话标记键换为 `rain_welcome_seen_v1`；命中「pure」主题时不再弹出
- 站点名改由 `siteConfig.title` 注入（`data-site-name`），不再硬编码
